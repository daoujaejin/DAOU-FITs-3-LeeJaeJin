typedef struct {
    char name[50];
    double investment_amount;
    double return_rate;
} Investment;

typedef struct {
    char name[50];
    int num_investments;
    Investment* investments[100];
} Customer;

void add_investment(Customer* customer, char* name, double investment_amount, double return_rate) {
    if (customer->num_investments < 100) {
        Investment* new_investment = (Investment*)malloc(sizeof(Investment));
        strcpy(new_investment->name, name);
        new_investment->investment_amount = investment_amount;
        new_investment->return_rate = return_rate;
        customer->investments[customer->num_investments] = new_investment;
        customer->num_investments++;
    } else {
        printf("투자 상품이 가득 찼습니다.\n");
    }
}

void calculate_and_print_profits(Customer* customer) {
    for (int i = 0; i < customer->num_investments; i++) {
        double profit = customer->investments[i]->investment_amount * customer->investments[i]->return_rate;
        printf("%s 투자 상품의 수익: %.2f\n", customer->investments[i]->name, profit);
    }
}

void free_investments(Customer* customer) {
    for (int i = 0; i < customer->num_investments; i++) {
        free(customer->investments[i]);
    }
}

typedef struct {
    char name[30];
    char account_number[10];
    struct tm start_date;
    int month;
    double deposit;
} FixedDeposit;

// 현재 날짜를 얻는 함수
struct tm get_current_date() {
    time_t now = time(NULL);
    struct tm* current_date = localtime(&now);
    return *current_date;
}

// 만기일 계산 함수
struct tm calculate_maturity_date(struct tm start_date, int month) {
    struct tm maturity_date = start_date;
    maturity_date.tm_mon += month;
    if (maturity_date.tm_mon > 11) {
        maturity_date.tm_year += maturity_date.tm_mon / 12;
        maturity_date.tm_mon %= 12;
    }
    return maturity_date;
}

// 두 날짜 비교 함수
int compare_dates(struct tm date1, struct tm date2) {
    if (date1.tm_year < date2.tm_year) return -1;
    if (date1.tm_year > date2.tm_year) return 1;
    if (date1.tm_mon < date2.tm_mon) return -1;
    if (date1.tm_mon > date2.tm_mon) return 1;
    if (date1.tm_mday < date2.tm_mday) return -1;
    if (date1.tm_mday > date2.tm_mday) return 1;
    return 0;
}

// 정기 예금을 만기일이 빠른 순으로 정렬하는 함수
void sort_by_maturity_date(FixedDeposit deposits[], int num_deposits) {
    for (int i = 0; i < num_deposits - 1; i++) {
        for (int j = i + 1; j < num_deposits; j++) {
            struct tm maturity_date1 = calculate_maturity_date(deposits[i].start_date, deposits[i].month);
            struct tm maturity_date2 = calculate_maturity_date(deposits[j].start_date, deposits[j].month);
            if (compare_dates(maturity_date1, maturity_date2) > 0) {
                FixedDeposit temp = deposits[i];
                deposits[i] = deposits[j];
                deposits[j] = temp;
            }
        }
    }
}

// 현재 날짜와 비교하여 만기 여부 확인하는 함수
bool check_maturity(FixedDeposit deposit) {
    struct tm current_date = get_current_date();
    struct tm maturity_date = calculate_maturity_date(deposit.start_date, deposit.month);
    return compare_dates(current_date, maturity_date) >= 0;
}

// 정렬된 정기 예금 리스트를 출력하는 함수
void display_deposits(FixedDeposit deposits[], int num_deposits) {
    printf("[ 정기 예금 목록 (만기일 빠른 순) ]\n");
    for (int i = 0; i < num_deposits; i++) {
        printf("%d. %s | 계좌: %s | 예금 시작: %04d-%02d-%02d | 만기: %04d-%02d-%02d | 금액: %.2f | ",
                i + 1,   
                deposits[i].name, deposits[i].account_number,
                deposits[i].start_date.tm_year + 1900, deposits[i].start_date.tm_mon + 1, deposits[i].start_date.tm_mday,
                calculate_maturity_date(deposits[i].start_date, deposits[i].month).tm_year + 1900,
                calculate_maturity_date(deposits[i].start_date, deposits[i].month).tm_mon + 1,
                deposits[i].month, deposits[i].deposit);
        if (check_maturity(deposits[i])) {
            printf("만기 도래 O |\n");
        } else {
            printf("만기 도래 X |\n");
        }
    }
}

typedef struct {
    char name[20];
    char ticker[6];
    int num;
    double closing_price[30];
    double average_price;
    double volatility;
} Stock;

double calculate_average(Stock st) {
    double res = 0;
    for (int i = 0; i < 30; i++)
        res += st.closing_price[i];
    return res / 30;
}

void calculate_volatility(Stock *st) {
    double avg = calculate_average(*st);
    double sum = 0;
    for (int i = 0; i < 30; i++) {
        sum += pow(st->closing_price[i] - avg, 2);
    }
    st->volatility = sqrt(sum / 30);
    st->average_price = avg;
}

int categorize_risk(Stock st) {
    int v = st.volatility;
    if (v <= 5) {
        return 0;
    } else if (v <= 10) {
        return 1;
    } else
        return 2;
}

void display_portfolio(Stock st) {
    printf("종목: %s (%s) | 보유 주식: %d | 변동성: %.2f | 위험 수준: ",
        st.name, st.ticker, st.num, st.volatility);
    int risk = categorize_risk(st);
    if (risk == 0) {
        printf("낮음\n");
    } else if (risk == 1) {
        printf("보통\n");
    } else {
        printf("높음\n");
    }
}
