int balance = 0;
char system_PW[5] = "1234";

bool check_PW() {
    static int total_error = 0;
    char pw[20];
    while (1) {
        printf("시스템 비밀번호를 입력하세요 :");
        scanf_s("%s", pw, 20);
        if (strncmp(pw, system_PW, 4) == 0)
            return true;
        if (++total_error == 3) {
            printf("비밀번호 오류 3회 발생\n");
            printf("신분증을 가지고 서비스센터를 방문하세요\n");
            return false;
        }
        printf("비밀번호가 틀렸습니다. 총 %d회 오류\n", total_error);
    }
}

void withdraw() {
    int wd;
    printf("출금하실 금액은? ");
    scanf_s("%d", &wd);
    if (balance < wd) {
        printf("잔액이 부족합니다. 현재 잔액은 %d원\n", balance);
    } else {
        balance -= wd;
        printf("출금 후 현재 잔액은 %d원\n", balance);
    }
}

void deposit() {
    int dep;
    printf("입금하실 금액은? ");
    scanf_s("%d", &dep);
    balance += dep;
    printf("입금 후 현재 잔액은 %d원\n", balance);
}

void check_balance() {
    printf("현재 잔액은 %d원\n", balance);
}
