#include <stdio.h>

typedef struct person
{
    char name[7];
    char account_number[10];
    double amount;
} PERSON;

void displayUsers(FILE* fp, int size) {
    PERSON user;
    puts("---------------------------------");
    puts("번호 이름      계좌번호  잔액");
    puts("---------------------------------");
    rewind(fp); // 파일의 시작위치로 이동
    for (int i = 0; i < size; i++)
    {
        fread(&user, sizeof(PERSON), 1, fp);
        printf("%3d %-8s %-3s %4.2f\n", i + 1, user.name, user.account_number, user.amount);
    }
    puts("---------------------------------");
 }
 void searchUser(FILE* fp, int size) {
    char searchName[7];
    PERSON user;
    int found = 0;
    printf("검색할 이름을 입력하세요: ");
    scanf("%s", searchName);
    puts("---------------------------------");
    puts("번호  이름   성별  나이");
    puts("---------------------------------");
    rewind(fp); // 파일의 시작위치로 이동
    for (int i = 0; i < size; i++)
    {
        fread(&user, sizeof(PERSON), 1, fp);
        if (strcmp(user.name, searchName) == 0) {
            printf("%3d %-8s %-3s %4.2f\n", i + 1, user.name, user.account_number, user.amount);
            found = 1;
        }
    }
    if (!found) {
        printf("사용자를 찾을 수 없습니다.\n");
    }
    puts("---------------------------------");
 }
void addUser(FILE* fp) {
    PERSON user;
    printf("\n이름 : "); scanf("%s", user.name);
    printf("계좌번호 : "); scanf("%s", user.account_number);
    printf("잔액 : "); scanf("%lf", &user.amount);
    fseek(fp, 0, SEEK_END); // 파일 끝으로 이동
    fwrite(&user, sizeof(PERSON), 1, fp);
    printf("새 고객이 삽입되었습니다.\n");
 }
void deleteUser(FILE* fp, int* size) {
    int no;
    PERSON* users = (PERSON*)malloc(*size * sizeof(PERSON));
    rewind(fp);
    fread(users, sizeof(PERSON), *size, fp);
    printf("삭제할 고객의 번호는? ");
    scanf("%d", &no);
    if (no < 1 || no > *size) {
        printf("잘못된 번호입니다.\n");
    }
    else {
        
        // 삭제 후 사용자 수 줄이기
        for (int i = no - 1; i < *size - 1; i++) {
            users[i] = users[i + 1];
        }
        (*size)--; // 사용자 수 감소
        // 파일을 새로 쓰기
        freopen("user.bin", "wb+", fp);
        fwrite(users, sizeof(PERSON), *size, fp);
        printf("사용자가 삭제되었습니다.\n");
    }
    free(users);
}

 void viewAllUsers(FILE* fp, int size) {
    if (size == 0) {
        printf("등록된 사용자가 없습니다.\n");
    }
    else {
        displayUsers(fp, size);
    }
 }
