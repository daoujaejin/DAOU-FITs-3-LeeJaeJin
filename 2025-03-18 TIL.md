typedef struct {
    double interest_rate;
    double loan_amount;
    double remaining_balance;
    int loan_period;  // 대출 기간 (년)
} Loan;

typedef struct {
    int account_number;
    char name[100];
    Loan* loans;  // 동적으로 할당된 대출 정보 배열
    int loan_count;
} Customer;

void pr_12_9_3() {
    int total;
    Customer **customers;

    printf("고객 수를 입력하세요: ");

        scanf("%d", &temp->account_number);
        printf("고객 %d가 받은 대출 수: ", i + 1);

        temp->loans = (Loan*)malloc(sizeof(Loan) * total);
        for (int j = 0; j < temp->loan_count; j++) {
            printf("대출 %d:\n", j + 1);
            printf("대출 금액을 입력하세요: ");

            scanf("%lf", &((temp->loans + j)->interest_rate));
            printf("대출 기간을 입력하세요(1년 이상): ");
            scanf("%d", &((temp->loans + j)->loan_period));
        }
        for (int j = 0; j < temp->loan_count; j++) {
            double ret;
            printf("상환할 금액을 입력하세요 (대출 %d): ", i + 1);
            scanf("%lf", &ret);
            (temp->loans + j)->remaining_balance = (temp->loans + j)->loan_amount - ret;
        }
        customers[i] = temp;
    }

    printf("입력된 고객 및 대출 정보:\n");
    for (int i = 0; i < total; i++) {
        printf("고객 %d: %s, 계좌번호: %d\n", i + 1, customers[i]->name, customers[i]->account_number);
        for (int j = 0; j < customers[i]->loan_count; j++) {

            j + 1, 
            customers[i]->loans[j].loan_amount,

            customers[i]->loans[j].loan_period, 
            customers[i]->loans[j].remaining_balance);
        }
        free(customers[i]);
    }
    free(customers);
}

        scanf("%s", customers[i].name);
        printf("고객 %d의 계좌 번호를 입력하세요: ", i + 1);
        scanf("%d", &customers[i].account);
        printf("고객 %d이 거래 내역 수: ", i + 1);
        scanf("%d", &customers[i].transactionCount);

        customers[i].transactions = (Transaction *) calloc(customers[i].transactionCount, sizeof(Transaction));
        for (int j = 0; j < customers[i].transactionCount; j++) {
            printf("거래 %d\n", j + 1);
            printf("주식 이름을 입력하세요: ");
            scanf("%s", customers[i].transactions[j].name);

            printf("거래 수량을 입력하세요: ");
            scanf("%d", &customers[i].transactions[j].amount);

            printf("거래 가격을 입력하세요: ");
            scanf("%lf", &customers[i].transactions[j].price);
        }
    }

    for (int i = 0; i < n; i++) {
        printf("고객 %d: %s, 계좌번호: %d\n", i + 1, customers[i].name, customers[i].account);
        for (int j = 0; j < customers[i].transactionCount; j++) {
            printf("거래 %d: 주식 이름: %s, 거래 수량: %d, 거래 가격: %.2lf\n",
                   j + 1,
                   customers[i].transactions[j].name,
                   customers[i].transactions[j].amount,
                   customers[i].transactions[j].price
            );
        }
    }

    for (int i = 0; i < n; i++) {
        free(customers[i].transactions);
    }
    free(customers);
}
