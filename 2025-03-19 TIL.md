void pr_12_13_1() {
    typedef struct {
        char stock_name[20];
        int quantity;
        double price;
    } StockOrder;

    int initial_orders;
    printf("초기 주문 개수를 입력하세요: ");
    scanf("%d", &initial_orders);
    StockOrder* orders = (StockOrder*)malloc(initial_orders * sizeof(StockOrder));
    
    for (int i = 0; i < initial_orders; i++) {
        printf("주식 주문 %d:\n", i + 1);
        printf("주식 이름: ");
        scanf("%s", orders[i].stock_name);
        printf("주식 수량: ");
        scanf("%d", &orders[i].quantity);
        printf("주식 가격: ");
        scanf("%lf", &orders[i].price);
    }

    printf("추가 주문을 하시겠습니까? (Y/N): ");
    char add_more;
    scanf(" %c", &add_more);
    if (add_more == 'Y' || add_more == 'y') {
        int add;
        printf("추가할 주문 개수를 입력하세요: ");
        scanf("%d", &add);
        for (int i = 0; i < add; i++) {
            // realloc을 사용하여 배열 크기 늘리기
            orders = (StockOrder*)realloc(orders, (initial_orders + 1) * sizeof(StockOrder));

            printf("주식 주문 %d:\n", initial_orders + 1);
            printf("주식 이름: ");
            scanf("%s", orders[initial_orders].stock_name);
            printf("주식 수량: ");
            scanf("%d", &orders[initial_orders].quantity);
            printf("주식 가격: ");
            scanf("%lf", &orders[initial_orders].price);

            initial_orders++;
        }
    }

    printf("주식 주문 내역:\n");
    for (int i = 0; i < initial_orders; i++) {
        printf("주식 이름: %s, 수량: %d, 가격: %.2f\n", orders[i].stock_name, orders[i].quantity, orders[i].price);
    }
}

void pr_12_16_1() {
    typedef enum {
        CASH,
        CARD,
        CRYPTO
    } TransactionType;

    typedef union {
        struct {
            double amount;
        } cash;
        struct {
            char card_number[20];
            int installment_months;
            double amount;
        } card;
        struct {
            char wallet_address[20];
            double amount;
        } crypto;
    } TransactionData;

    typedef struct {
        int id;
        TransactionType type;
        char customer_name[20];
        TransactionData data;
    } Transaction;

    Transaction *transactions = NULL;
    int n = 0;

    while (1) {
        int choice;
        printf("1. 거래 추가\n");
        printf("2. 거래 내역 출력\n");
        printf("3. 종료\n");
        printf("선택: ");
        scanf("%d", &choice);
        if (choice == 1) {
            transactions = (Transaction*)realloc(transactions, (n + 1) * sizeof(Transaction));

            (transactions)[n].id = n + 1;
            printf("거래 유형을 선택하세요 (0: 현금, 1: 카드, 2: 가상화폐): ");
            int type;
            scanf("%d", &type);
            (transactions)[n].type = (TransactionType)type;

            printf("고객 이름을 입력하세요: ");
            scanf("%s", (transactions)[n].customer_name);

            switch ((transactions)[n].type) {
                case CASH:
                    printf("거래 금액을 입력하세요: ");
                    scanf("%lf", &(transactions)[n].data.cash.amount);
                    break;
                case CARD:
                    printf("카드 번호를 입력하세요: ");
                    scanf("%19s", (transactions)[n].data.card.card_number);
                    printf("할부 개월 수를 입력하세요: ");
                    scanf("%d", &(transactions)[n].data.card.installment_months);
                    printf("거래 금액을 입력하세요: ");
                    scanf("%lf", &(transactions)[n].data.card.amount);
                    break;
                case CRYPTO:
                    printf("지갑 주소를 입력하세요: ");
                    scanf("%49s", (transactions)[n].data.crypto.wallet_address);
                    printf("거래 금액을 입력하세요: ");
                    scanf("%lf", &(transactions)[n].data.crypto.amount);
                    break;
            }
            n++;
        } else if (choice == 2) {
            for (int i = 0; i < n; i++) {
                printf("거래 ID: %d, 거래 유형: ", transactions[i].id);
                switch (transactions[i].type) {
                    case CASH:
                        printf("현금 거래, ");
                        printf("고객 이름: %s, 금액: %.2f\n", transactions[i].customer_name, transactions[i].data.cash.amount);
                        break;
                    case CARD:
                        printf("카드 거래, ");
                        printf("고객 이름: %s, 카드 번호: %s, 할부 개월 수: %d, 금액: %.2f\n", 
                            transactions[i].customer_name, 
                            transactions[i].data.card.card_number, 
                            transactions[i].data.card.installment_months,
                            transactions[i].data.card.amount);
                        break;
                    case CRYPTO:
                        printf("가상화폐 거래, ");
                        printf("고객 이름: %s, 지갑 주소: %s, 금액: %.2f\n", 
                            transactions[i].customer_name, 
                            transactions[i].data.crypto.wallet_address, 
                            transactions[i].data.crypto.amount);
                        break;
                }
            }
        } else {
            return ;
        }
    }

void lec_13_bitCal() {
    typedef struct {
        uint8_t flags;
    } UserPermissions;
    
    #define READ_MASK       0x01
    #define WRITE_MASK      0x02
    #define EXECUTE_MASK    0x04
    #define ADMIN_MASK      0x08

    UserPermissions user1;
    user1.flags = 0;

    int choice;

    printf("사용자 권한을 설정합니다.\n");
    printf("READ 권한을 부여할까요? (1: 예, 0: 아니오): ");
    scanf("%d", &choice);
    if (choice)
        user1.flags |= READ_MASK;
    printf("WRITE 권한을 부여할까요? (1: 예, 0: 아니오): ");
    scanf("%d", &choice);
    if (choice)
        user1.flags |= WRITE_MASK;
    printf("EXECUTE 권한을 부여할까요? (1: 예, 0: 아니오): ");
    scanf("%d", &choice);
    if (choice)
        user1.flags |= EXECUTE_MASK;
    printf("ADMIN 권한을 부여할까요? (1: 예, 0: 아니오): ");
    scanf("%d", &choice);
    if (choice)
        user1.flags |= ADMIN_MASK;

    printf("\n사용자 권한 상태:\n");
    printf("- READ: %c\n", user1.flags & READ_MASK ? 'O' : 'X');
    printf("- WRITE: %c\n", user1.flags & WRITE_MASK ? 'O' : 'X');
    printf("- EXECUTE: %c\n", user1.flags & EXECUTE_MASK ? 'O' : 'X');
    printf("- ADMIN: %c\n", user1.flags & ADMIN_MASK ? 'O' : 'X');

    printf("\nUserPermissions 구조체 크기: %d 바이트\n", sizeof(user1));
}
