void addUser(FILE* fp) {
    PERSON user;
    printf("\n이름 : "); scanf("%s", user.name);
    printf("성별 : "); scanf("%s", user.gender);
    printf("나이 : "); scanf("%d", &user.age);
    fseek(fp, 0, SEEK_END); // 파일 끝으로 이동
    fwrite(&user, sizeof(PERSON), 1, fp);
    printf("새 사용자가 삽입되었습니다.\n");
 }
void deleteUser(FILE* fp, int* size) {
    int no;
    PERSON* users = (PERSON*)malloc(*size * sizeof(PERSON));
    rewind(fp);
    fread(users, sizeof(PERSON), *size, fp);
    printf("삭제할 사용자의 번호는? ");
    scanf("%d", &no);
    if (no < 1 || no > *size) {
        printf("잘못된 번호입니다.\n");
    }
    else {
        // 삭제 후 사용자 수 줄이기
        for (int i = no - 1; i < *size - 1; i++) {
            users[i] = users[i + 1];
        }
        (*size)--; // 사용자 수 감소
        // 파일을 새로 쓰기
        freopen("user.bin", "wb", fp);
        fwrite(users, sizeof(PERSON), *size, fp);
        printf("사용자가 삭제되었습니다.\n");
    }
    free(users);
}

    if (size == 0) {

    }
    else {
        displayUsers(fp, size);
    }
 }

 // 고객 데이터를 JSON 파일로 저장
void saveToJson(FILE* fp, int size) {
    CUSTOMER customer;
    FILE* jsonFile = fopen("C:\\cstudy\\C_Study\\customers.json", "w");
    if (jsonFile == NULL) {
        printf("JSON 파일을 생성할 수 없습니다.\n");
        return;
    }
    fprintf(jsonFile, "[\n");  // JSON 배열 시작
    rewind(fp); // 파일 처음으로 이동
    for (int i = 0; i < size; i++) {
        fread(&customer, sizeof(CUSTOMER), 1, fp);
        fprintf(jsonFile, "  {\n");
        fprintf(jsonFile, "    \"name\": \"%s\",\n", customer.name);
        fprintf(jsonFile, "    \"accountNo\": \"%s\",\n", customer.accountNo);
        fprintf(jsonFile, "    \"balance\": %.2f\n", customer.balance);
        if (i == size - 1)
            fprintf(jsonFile, "  }\n");  // 마지막 요소 (쉼표 없음)
        else
            fprintf(jsonFile, "  },\n"); // 쉼표 추가
    }
    fprintf(jsonFile, "]\n"); // JSON 배열 종료
    fclose(jsonFile);
    printf("고객 정보가 customers.json 파일에 저장되었습니다.\n");
 }

        fp = fopen("C:\\cstudy\\C_Study\\transactions.bin", "wb+");
        if (fp == NULL) {
            printf("파일 생성 오류\n");
            return;
        }
        createDefaultC(fp);
    }
    // 파일 크기 확인
    fseek(fp, 0, SEEK_END);
    size = ftell(fp) / sizeof(C);
    if (size == 0) {
        printf("저장된 고객 정보가 없습니다.\n");
        fclose(fp);
        return;
    }
    // JSON 파일로 변환
    saveToJsonC(fp, size);
    
    fclose(fp);
    return;
 }
