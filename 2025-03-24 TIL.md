// 날짜 연산을 위한 함수
void add_months(struct tm* date, int months) {
    date->tm_mon += months;
    mktime(date);
 }
 // 정기 예금 정렬 (만기일이 가까운 순)
 void sort_by_maturity_date(FixedDeposit deposits[], int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            struct tm d1 = deposits[j].start_date;
            struct tm d2 = deposits[j + 1].start_date;
            add_months(&d1, deposits[j].duration_months);
            add_months(&d2, deposits[j + 1].duration_months);
            if (mktime(&d1) > mktime(&d2)) {
                FixedDeposit temp = deposits[j];
                deposits[j] = deposits[j + 1];
                deposits[j + 1] = temp;
            }
        }
    }
 }
 // 정기 예금 만기 확인
void check_maturity(FixedDeposit* deposit) {
    time_t now = time(NULL);
    struct tm maturity_date = deposit->start_date;
    add_months(&maturity_date, deposit->duration_months);
    if (difftime(mktime(&maturity_date), now) <= 0) {
        printf("만기 도래 ✅ | ");
    }
    else {
        printf("만기 대기 ￿ | ");
    }
 }
 // 정기 예금 목록 출력
void display_deposits(FixedDeposit deposits[], int n) {
    printf("\n[ 정기 예금 목록 (만기일 빠른 순) ]\n");
    for (int i = 0; i < n; i++) {
        struct tm maturity_date = deposits[i].start_date;
        add_months(&maturity_date, deposits[i].duration_months);
        printf("%d. %s | 계좌: %s | 예금 시작: %04d-%02d-%02d | 만기: %04d-%02d-%02d | 금액: %.2lf | ",
            i + 1, deposits[i].name, deposits[i].account_number,
            deposits[i].start_date.tm_year + 1900, deposits[i].start_date.tm_mon + 1, 
deposits[i].start_date.tm_mday,
            maturity_date.tm_year + 1900, maturity_date.tm_mon + 1, 
maturity_date.tm_mday,
            deposits[i].amount);
        check_maturity(&deposits[i]);
        printf("\n");
    }
 }
